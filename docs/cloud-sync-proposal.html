1. Problem Statement

User data (favorites, error book, SRS progress, stats, badges) lives in browser localStorage. Switching devices, clearing browser data, or using a different browser loses everything. The app needs to support 100 families with ~200 kids, with up to 20 users online simultaneously.

2. Design Principles

PrincipleRationale
Offline-firstApp works without network. Sync is opportunistic, not blocking. Network is unreliable in schools.
Layered architectureRoutes → Services → Database. Each layer testable in isolation. New features = new folder.
Environment-driven configNo hardcoded values. Same code runs in dev/test/prod with different env vars.
Structured error handlingEvery error returns the same JSON shape. Logs are machine-parseable.
Operation-based syncArray fields use ops (add/remove) instead of last-write-wins. Multi-device edits merge correctly.
Zero-downtime extensibilityAPI versioning, schema migrations, feature flags. Add features without breaking clients.
Kid-friendly authNo email. Invite code + name + 4-digit PIN. Memorable for 6-year-olds.

7.3 Migration Runner

# server/database.py (excerpt)
import sqlite3
from pathlib import Path

def get_connection(db_path: str) -> sqlite3.Connection:
    conn = sqlite3.connect(db_path, check_same_thread=False)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")
    conn.execute("PRAGMA synchronous=NORMAL")
    conn.execute("PRAGMA busy_timeout=5000")
    conn.execute("PRAGMA foreign_keys=ON")
    return conn

def run_migrations(conn: sqlite3.Connection, migrations_dir: Path):
    """Apply pending SQL migrations in order."""
    conn.execute("""CREATE TABLE IF NOT EXISTS schema_version (
        version INTEGER PRIMARY KEY, name TEXT, applied_at TEXT DEFAULT (datetime('now'))
    )""")
    applied = {row[0] for row in conn.execute("SELECT version FROM schema_version")}
    migration_files = sorted(migrations_dir.glob("*.sql"))

    for mf in migration_files:
        version = int(mf.name.split("_")[0])
        if version in applied:
            continue
        sql = mf.read_text()
        conn.executescript(sql)
        conn.execute("INSERT INTO schema_version (version, name) VALUES (?, ?)",
                     (version, mf.name))
        conn.commit()
        print(f"  Applied migration: {mf.name}")
Extensibility: To add a new feature (e.g., leaderboard), just create 004_add_leaderboard.sql in the migrations folder. On next server start, it auto-applies. No manual SQL needed.

8.3 Token Design

# Token structure: base64(JSON payload) + "." + HMAC signature
# Payload: {"pid": "uuid", "fid": "uuid", "exp": "2026-08-28T10:00:00Z"}
# Signature: HMAC-SHA256(payload_bytes, server_secret)[:32]

import json, base64, hmac, hashlib
from datetime import datetime, timedelta

def create_token(profile_id: str, family_id: str, secret: str, days: int = 30) -> str:
    payload = {
        "pid": profile_id,
        "fid": family_id,
        "exp": (datetime.utcnow() + timedelta(days=days)).isoformat()
    }
    payload_b64 = base64.urlsafe_b64encode(json.dumps(payload).encode()).decode()
    sig = hmac.new(secret.encode(), payload_b64.encode(), hashlib.sha256).hexdigest()[:32]
    return f"{payload_b64}.{sig}"

def verify_token(token: str, secret: str) -> dict:
    payload_b64, sig = token.rsplit(".", 1)
    expected = hmac.new(secret.encode(), payload_b64.encode(), hashlib.sha256).hexdigest()[:32]
    if not hmac.compare_digest(sig, expected):
        raise AuthError("invalid_token")
    payload = json.loads(base64.urlsafe_b64decode(payload_b64))
    if datetime.fromisoformat(payload["exp"]) < datetime.utcnow():
        raise AuthError("token_expired")
    return payload

8.4 Rate Limiting

# server/auth/rate_limiter.py
# Sliding window: max N attempts per IP per window

class RateLimiter:
    def __init__(self, max_attempts: int = 5, window_seconds: int = 60):
        self.max_attempts = max_attempts
        self.window_seconds = window_seconds

    def check(self, conn: Connection, ip: str):
        cutoff = (datetime.utcnow() - timedelta(seconds=self.window_seconds)).isoformat()
        count = conn.execute(
            "SELECT COUNT(*) FROM login_attempts WHERE ip_address=? AND attempted_at>?",
            (ip, cutoff)
        ).fetchone()[0]
        if count >= self.max_attempts:
            raise RateLimitExceeded()

    def record(self, conn: Connection, ip: str):
        conn.execute("INSERT INTO login_attempts (ip_address) VALUES (?)", (ip,))
        # Cleanup old entries
        cutoff = (datetime.utcnow() - timedelta(seconds=self.window_seconds * 2)).isoformat()
        conn.execute("DELETE FROM login_attempts WHERE attempted_at < ?", (cutoff,))
        conn.commit()

10. Sync Engine

10.1 Two-Layer Strategy

Layer 1 — State snapshot (all keys): Last-write-wins per key. Simple, handles stats/badges/spacedRep where the whole object is the unit.

Layer 2 — Operation log (array keys only): For favorites and errorBook, track individual add/remove operations. On conflict, replay ops in timestamp order to produce the merged result.

10.2 Operation-Based Merge (favorites & errorBook)

# server/sync/operations.py

def merge_array_ops(server_ops: list, client_ops: list, server_state: list) -> list:
    """
    Merge operations from both server and client into a final state.

    Algorithm:
    1. Combine all ops, sort by timestamp
    2. Replay from empty set:
       - 'add' -> include item
       - 'remove' -> exclude item
    3. Return final list (deduplicated, ordered by first-add time)
    """
    all_ops = sorted(server_ops + client_ops, key=lambda o: o['at'])

    included = {}   # char -> first_add_timestamp
    excluded = set()

    for op in all_ops:
        target = op['target']
        if op['op'] == 'add':
            if target not in excluded:
                if target not in included:
                    included[target] = op['at']
        elif op['op'] == 'remove':
            excluded.add(target)
            included.pop(target, None)

    # Return in order of first addition
    return sorted(included.keys(), key=lambda c: included[c])

11. TTS Service (with caching)

11.1 Architecture

# server/tts/cache.py

import hashlib
from pathlib import Path

class TTSCache:
    def __init__(self, cache_dir: str, enabled: bool = True):
        self.cache_dir = Path(cache_dir)
        self.cache_dir.mkdir(exist_ok=True)
        self.enabled = enabled

    def get(self, text: str) -> bytes | None:
        if not self.enabled:
            return None
        path = self._path(text)
        return path.read_bytes() if path.exists() else None

    def put(self, text: str, audio: bytes):
        if not self.enabled:
            return
        self._path(text).write_bytes(audio)

    def _path(self, text: str) -> Path:
        key = hashlib.sha256(text.encode()).hexdigest()[:16]
        return self.cache_dir / f"{key}.mp3"

    @property
    def size_mb(self) -> float:
        return sum(f.stat().st_size for f in self.cache_dir.iterdir()) / 1_048_576

12.1 SyncService Module

// js/services.js (addition)

const SyncService = (() => {
  const SYNC_KEYS = ['favorites', 'errorBook', 'stats', 'badges', 'spacedRep', 'favContext'];
  const OP_KEYS = ['favorites', 'errorBook']; // keys that use operation-based merge
  let _debounceTimer = null;
  let _token = null;
  let _enabled = false;
  let _pendingOps = [];  // queued operations for next push

  // --- Public API ---
  function isEnabled() { return _enabled && !!_token; }

  function loadToken() {
    _token = localStorage.getItem('szb_syncToken');
    _enabled = !!_token;
    _pendingOps = JSON.parse(localStorage.getItem('szb_pendingOps') || '[]');
  }

  async function register(inviteCode, name, pin, avatar) {
    const res = await _fetch('/api/v1/auth/register', 'POST', {
      invite_code: inviteCode, display_name: name, pin, avatar
    });
    if (res.ok) _setToken(res.data.token);
    return res;
  }

  async function login(inviteCode, name, pin) {
    const res = await _fetch('/api/v1/auth/login', 'POST', {
      invite_code: inviteCode, display_name: name, pin
    });
    if (res.ok) _setToken(res.data.token);
    return res;
  }

  async function pull() {
    if (!isEnabled()) return;
    try {
      _updateIndicator('syncing');
      const res = await _fetch('/api/v1/sync/pull', 'GET');
      if (!res.ok) { _handleAuthError(res); return; }

      const localTs = _getTimestamps();
      for (const key of SYNC_KEYS) {
        if (!res.data[key]) continue;
        const serverTime = res.data[key].updated_at;
        const localTime = localTs[key] || '1970-01-01';
        if (serverTime > localTime) {
          State.set(key, res.data[key].value);
          State.persist(key);
          localTs[key] = serverTime;
        }
      }
      _saveTimestamps(localTs);
      _updateIndicator('synced');
    } catch (e) {
      _updateIndicator('offline');
    }
  }

  function recordOp(key, op, target, data = null) {
    if (!OP_KEYS.includes(key)) return;
    _pendingOps.push({ key, op, target, data, at: new Date().toISOString() });
    localStorage.setItem('szb_pendingOps', JSON.stringify(_pendingOps));
  }

  function markDirty(key) {
    if (!isEnabled()) return;
    const ts = _getTimestamps();
    ts[key] = new Date().toISOString();
    _saveTimestamps(ts);
    clearTimeout(_debounceTimer);
    _debounceTimer = setTimeout(() => _push(), 2000);
  }

  // --- Private ---
  async function _push() { /* collect state + ops, POST to /push, handle conflicts */ }
  function _setToken(token) { _token = token; _enabled = true; localStorage.setItem('szb_syncToken', token); }
  function _getTimestamps() { return JSON.parse(localStorage.getItem('szb_syncTs') || '{}'); }
  function _saveTimestamps(ts) { localStorage.setItem('szb_syncTs', JSON.stringify(ts)); }
  function _updateIndicator(s) { /* update header icon */ }
  function _handleAuthError(res) { if (res.error === 'token_expired') { /* show re-login */ } }
  async function _fetch(path, method, body) { /* wrapper with auth header */ }

  return { isEnabled, loadToken, register, login, pull, recordOp, markDirty };
})();

// sw.js — enhanced for offline sync

const CACHE_NAME = 'hanzigo-v2';
const STATIC_ASSETS = [
  '/', '/index.html', '/css/style.css',
  '/js/constants.js', '/js/state.js', '/js/data.js',
  '/js/services.js', '/js/ui.js', '/js/controllers.js',
  '/config.json', '/manifest.json'
];

// Install: cache all static assets
self.addEventListener('install', (e) => {
  e.waitUntil(caches.open(CACHE_NAME).then(c => c.addAll(STATIC_ASSETS)));
});

// Fetch: serve from cache, fall back to network
self.addEventListener('fetch', (e) => {
  const url = new URL(e.request.url);

  // API calls: always network (sync service handles offline)
  if (url.pathname.startsWith('/api/')) return;

  // TTS: cache-first (audio doesn't change)
  if (url.pathname === '/tts') {
    e.respondWith(caches.match(e.request).then(r => r || fetch(e.request).then(res => {
      const clone = res.clone();
      caches.open('hanzigo-tts').then(c => c.put(e.request, clone));
      return res;
    })));
    return;
  }

  // Static: cache-first, network fallback
  e.respondWith(caches.match(e.request).then(r => r || fetch(e.request)));
});

// Background Sync: retry failed pushes when online
self.addEventListener('sync', (e) => {
  if (e.tag === 'sync-push') {
    e.waitUntil(/* read pendingOps from IndexedDB, POST to server */);
  }
});

20. Migration Path (Existing Users)

Scenario: Users already have localStorage data from the current version (no sync). They upgrade to the sync-enabled version.

  1. On first launch of new version, app detects existing localStorage data but no sync token
  2. App works normally (local mode). A subtle banner appears: "\u2601\ufe0f \u65b0\u529f\u80fd\uff1a\u4e91\u540c\u6b65\uff01\u6362\u8bbe\u5907\u4e0d\u4e22\u6570\u636e"
  3. User taps banner or opens settings → sees sync setup
  4. On register/login, ALL existing localStorage data is pushed to server as initial state
  5. From that point on, sync runs automatically

No data is ever lost. Users who never enable sync continue using the app exactly as before.
